// Refreshed forms to align with label-driven updates.
// Use "..." to denote a block body captured between the opening directive and END.

INPUT
INPUT <label>

SEND <label>  // shorthand for SEND <label> MESSAGE INPUT
SEND <label> MESSAGE <value>  // send a single message to an agent
SEND <label> MESSAGE ... END  // send message in block
SEND <label> LIST <value>  // send multiple messages to an agent
SEND <label> LIST ... END   // send multiple messages in the list block

SEND <label> IN <label>  // shorthand for SEND <label> IN <label> MESSAGE INPUT
SEND <label> IN <label> MESSAGE <value>  // send a single message to an agent in a library
SEND <label> IN <label> MESSAGE ... END  // send message in block
SEND <label> IN <label> LIST <value>  // send multiple messages to an agent in a library
SEND <label> IN <label> LIST ... END   // send multiple messages in the list block

FACT <label> IN <label>
FACT <label>

IF <predicate> THEN <value> ELSE <value>  // statement does not need END, like a block
IF <predicate> THEN ... END
ELSE IF <predicate> THEN ... END
ELSE ... END

LIST <value>              // read bullets, write bullets
LIST <value> FROM BULLETS // explicit synonym for the default
LIST <value> FROM LINES   // read lines (single newline), write bullets
LIST <value> FROM PARAGRAPHS // read paragraphs (blank line separated), write bullets

LIST <value> AS BULLETS // write bullets (default)
LIST <value> AS LINES   // read bullets, write single-line entries
LIST <value> AS PARAGRAPHS // read bullets, write paragraphs

LET <label> BE <value>
LET <label> BE ... END

USE <label>
